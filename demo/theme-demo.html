(() => {
  const root = document.querySelector('[data-frost-root]');

  const button = event.currentTarget;
  button.disabled = true;

  button.textContent = '保存中…';
  // 仅用于展示等待状态；业务项目应 await 实际请求并处理失败。
  await new Promise(resolve => setTimeout(resolve, 700));
  button.disabled = false;
  button.removeAttribute('aria-busy');
  button.textContent = '保存变更';
  scope.querySelector('[data-example-feedback]').textContent = '成功：本地操作演示完成，没有发送网络请求。';

<script>
(() => {
  const root = document.querySelector('[data-frost-root]');
  const frost = Frost.init(root);
const scope = root.querySelector('#example-inputs');
const form = scope.querySelector('form');
const name = scope.querySelector('#demo-name');
const error = scope.querySelector('#demo-name-error');
const result = scope.querySelector('[data-example-result]');
const validate = () => {
  const valid = name.value.trim().length >= 2 && name.value.trim().length <= 40;
  name.setAttribute('aria-invalid', String(!valid));
  error.textContent = valid ? '' : '错误：请填写 2–40 个字符的项目名称，纯空格不能作为名称。修正后可以再次保存；已填写的其他内容会保留。';
  return valid;
};
name.addEventListener('input', () => { if (name.getAttribute('aria-invalid') === 'true') validate(); });
form.addEventListener('submit', event => {
  event.preventDefault();
  if (!validate()) { result.textContent = '尚未保存，请先修正项目名称。'; name.focus(); return; }
  if (!form.reportValidity()) return;
  result.textContent = '成功：本地示例“' + name.value.trim() + '”已校验，内容未上传或持久化。';
});
const search = scope.querySelector('#demo-search');
search.addEventListener('input', () => {
  const term = search.value.trim().toLowerCase();
  const names = ['按钮','输入框','文本域','选择框','复选框','单选框','开关','卡片','标签','Tabs','折叠面板','下拉菜单','行内提示','Toast','对话框','加载状态','空状态'];
  const matches = names.filter(value => value.toLowerCase().includes(term));
  scope.querySelector('#demo-search-result').textContent = !term ? '输入名称，查看本页组件匹配结果。' : matches.length ? '找到：' + matches.join('、') : '没有匹配项，请尝试其他名称。';
});
form.addEventListener('reset', () => {
  name.removeAttribute('aria-invalid'); error.textContent = '';
  const password = scope.querySelector('#demo-password');
  const toggle = scope.querySelector('[data-frost-password]');
  password.type = 'password'; toggle.setAttribute('aria-pressed', 'false');
  toggle.setAttribute('aria-label', '显示密码'); toggle.querySelector('[data-frost-password-label]').textContent = '显示';
  result.textContent = '已重置，内容未保存。';
  scope.querySelector('#demo-search-result').textContent = '输入名称，查看本页组件匹配结果。';
});
})();
</script>

11 / ACCORDION

按需展开帮助、说明或次要内容。

流光什么时候出现？

默认就有完整的静态渐变边框。聚焦后开始流动，失焦后停在当前位置；错误、禁用和只读时使用对应的普通边界。系统要求减少动态效果时，渐变保持静态。

如何复用到现有项目？

加载 frost.css 与 frost.js，在容器上添加 data-frost-root，然后调用 Frost.init(root)。主题只保留浅色和深色。

open 属性由浏览器管理。
